function(restore_lib name)
 file(GLOB src ${ARGN})
 foreach(file IN LISTS src)
  get_filename_component(dir "${file}" DIRECTORY)
  list(APPEND original_dirs "${dir}")
 endforeach()
 if(ANDROID)
  set(ported)
  foreach(file IN LISTS src)
   file(READ "${file}" content)
   string(REPLACE "CURLOPT_SSL_VERIFYPEER, 0" "CURLOPT_SSL_VERIFYPEER, 1L" content "${content}")
   string(REPLACE "http://gs.apple.com" "https://gs.apple.com" content "${content}")
   get_filename_component(base "${file}" NAME)
   set(dest "${CMAKE_CURRENT_BINARY_DIR}/ported/${name}/${base}")
   file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/ported/${name}")
   file(WRITE "${dest}" "${content}")
   list(APPEND ported "${dest}")
   get_filename_component(dir "${file}" DIRECTORY)
   list(APPEND original_dirs "${dir}")
  endforeach()
  set(src ${ported})
 endif()
 add_library(${name} STATIC ${src})
 target_include_directories(${name} PRIVATE ${original_dirs})
 target_include_directories(${name} PUBLIC ${RESTORE_INCLUDES})
 if(ANDROID)
 target_compile_definitions(${name} PRIVATE curl_easy_init=menhir_restore_curl_init)
endif()
 target_compile_definitions(${name} PRIVATE HAVE_CONFIG_H PACKAGE_VERSION="menhir" PACKAGE_STRING="menhir" PACKAGE_NAME="${name}")
 target_link_libraries(${name} PUBLIC menhir_http zip)
endfunction()
