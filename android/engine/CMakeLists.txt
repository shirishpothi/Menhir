cmake_minimum_required(VERSION 3.22)
project(menhir_engine C ASM)
if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Use build.sh: Android arm64-v8a is required")
endif()
set(CMAKE_C_STANDARD 11)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
execute_process(COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/prepare-tests.py ${CMAKE_CURRENT_BINARY_DIR} COMMAND_ERROR_IS_FATAL ANY)
option(MENHIR_WITH_NET "Enable services networking" OFF)
option(MENHIR_WITH_RESTORE "Enable restore services" OFF)
option(MENHIR_WITH_AUDIO "Enable audio services" OFF)
file(GLOB CORE CONFIGURE_DEPENDS "${ROOT}/core/*.c")
add_library(menhir-engine STATIC ${CORE} ${ROOT}/host/jit_signal.c ${ROOT}/host/jit_region.c ${ROOT}/host/enter.S ${ROOT}/host/disk.c ${ROOT}/host/compat_android.c)
target_include_directories(menhir-engine PUBLIC ${ROOT}/core ${ROOT}/host ${ROOT}/devices)
target_compile_options(menhir-engine PRIVATE -Wall -Wextra -ffixed-x18)
set_source_files_properties(${ROOT}/core/mmu.c PROPERTIES COMPILE_OPTIONS -mgeneral-regs-only)
target_link_libraries(menhir-engine PUBLIC m dl)
enable_testing()
foreach(t classify jit_region operands mmu shutdown jit mem exc devices)
  set(src ${CMAKE_CURRENT_BINARY_DIR}/test_${t}.c)
  if(t STREQUAL "classify" OR t STREQUAL "operands")
    if(t STREQUAL "classify")
      set(asm classify_cases.s)
    else()
      set(asm operand_cases.s)
    endif()
    list(APPEND src ${CMAKE_CURRENT_BINARY_DIR}/${asm}.S)
  elseif(t STREQUAL "jit" OR t STREQUAL "mem" OR t STREQUAL "exc" OR t STREQUAL "devices")
    list(APPEND src ${CMAKE_CURRENT_BINARY_DIR}/${t}_programs.S ${CMAKE_CURRENT_BINARY_DIR}/harness.S)
  endif()
  add_executable(test_${t} ${src} test-host.c)
  target_include_directories(test_${t} PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ${ROOT}/tests)
  target_compile_options(test_${t} PRIVATE -ffixed-x18 -march=armv8.4-a+crypto -UNDEBUG)
  target_link_libraries(test_${t} PRIVATE menhir-engine)
  add_test(NAME ${t} COMMAND test_${t})
endforeach()

file(GLOB TLS_SRC "${ROOT}/third_party/mbedtls/library/*.c")
add_library(engine-crypto STATIC ${TLS_SRC})
target_include_directories(engine-crypto PUBLIC ${ROOT}/third_party/mbedtls/include)
set(DEVICE_NAMES pvg_lite pvg_null gicv3 timer pl011 pl031 cfg bdif pcie virtio virtio_blk virtio_net virtio_input virtio_snd virtio_usb virtio_balloon xhci vmapple aes pl061)
foreach(d IN LISTS DEVICE_NAMES)
  target_sources(menhir-engine PRIVATE ${ROOT}/devices/${d}.c)
endforeach()
target_link_libraries(menhir-engine PUBLIC engine-crypto)
add_executable(hengevm main.c)
target_link_libraries(hengevm PRIVATE menhir-engine)
if(MENHIR_WITH_NET OR MENHIR_WITH_RESTORE OR MENHIR_WITH_AUDIO)
  set(MENHIR_SERVICES_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(${ROOT}/android/deps services)
  if(MENHIR_WITH_NET)
    target_link_libraries(menhir-engine PUBLIC menhir::net)
  endif()
  if(MENHIR_WITH_RESTORE)
    target_link_libraries(menhir-engine PUBLIC menhir::restore)
  endif()
  if(MENHIR_WITH_AUDIO)
    target_link_libraries(menhir-engine PUBLIC menhir::audio)
  endif()
endif()
add_executable(test_platform test-platform.c)
target_link_libraries(test_platform PRIVATE menhir-engine)
add_test(NAME platform COMMAND test_platform)
